% RevTeX 4.2f: check a deferred float only after an eligible-width pass.
% This preserves layout and genuine stuck-float detection in the column pass.
\makeatletter
% Full-width floats cannot pass an earlier column float of the same type.
% Check the queue head, rather than treating a later wide float as eligible.
% The ordinary column pass still diagnoses a truly unplaceable column float.
\let\manuscript@check@deferlist@stuck\check@deferlist@stuck
\def\check@deferlist@stuck#1{%
  \ifx#1\do@startpage
    \begingroup
    \def\manuscript@next{\endgroup}%
    \def\@elt##1{%
      \ifdim\wd##1<\textwidth\else
        \def\manuscript@next{\endgroup\manuscript@check@deferlist@stuck#1}%
      \fi
      \let\@elt\@gobble}%
    \@deferlist
    \manuscript@next
  \else
    \manuscript@check@deferlist@stuck#1%
  \fi}
\makeatother

% Keep at most one marked vertical stack on each physical page. Other floats
% and text retain normal flow; a second stack waits for the next page.
\makeatletter
\newcount\manuscript@stackpage
\manuscript@stackpage=-1
\newcommand{\StackedFigure}{%
  \expandafter\gdef\csname manuscript@stack@\number\@currbox\endcsname{1}}
% Float box registers are reused, so clear the previous marker on allocation.
\let\manuscript@xfloat\@xfloat
\def\@xfloat#1[#2]{%
  \manuscript@xfloat#1[#2]%
  \expandafter\global\expandafter\let
    \csname manuscript@stack@\number\@currbox\endcsname\relax}
\def\manuscript@ifstack#1{%
  \expandafter\ifx\csname manuscript@stack@\number#1\endcsname\relax
    \expandafter\@secondoftwo
  \else
    \expandafter\@firstoftwo
  \fi}
\def\manuscript@claimstack#1{%
  \manuscript@ifstack#1{\global\manuscript@stackpage=\c@page}{}%
}
\let\manuscript@addtotoporbot\@addtotoporbot
\def\@addtotoporbot{%
  \manuscript@ifstack\@currbox{%
    \ifnum\manuscript@stackpage=\c@page
      \@insertfalse
    \else
      \manuscript@addtotoporbot
      \if@insert\manuscript@claimstack\@currbox\fi
    \fi
  }{\manuscript@addtotoporbot}}
% Apply the same quota when a barrier or document end drains float-only pages.
\let\manuscript@test@colfloat\test@colfloat
\def\test@colfloat#1{%
  \manuscript@test@colfloat#1%
  \manuscript@ifstack#1{%
    \ifnum\manuscript@stackpage=\c@page\@testtrue\fi
  }{}}
\let\manuscript@wtryfc\@wtryfc
\def\@wtryfc#1{%
  \manuscript@claimstack#1%
  \manuscript@wtryfc#1}
\makeatother
